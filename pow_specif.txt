/**
 * The pow subroutine calculates powers
 * of non-negative bases and exponents.
 *
 * Arguments:
 *  
 *      base - the exp base
 *      exp - the exponent
 *
 * Return value: 'base' raised to the power of 'exp'

 int pow(int base, int exp) {
        int total = 1;
        while(exp) {
            base *= base;
            exp--;
        }
        total = base;
        return total;
 }
